<ResponseField name="id" type="string">
  The player ID that plays & stats refer to.
</ResponseField>
<ResponseField name="teamId" type="string">
  The player's team, a key of `teams`.
</ResponseField>
<ResponseField name="name" type="string">
  The player's full name.
</ResponseField>
<ResponseField name="shortName" type="string">
  The name play descriptions use, such as `B.Purdy`.
</ResponseField>
<ResponseField name="jersey" type="string">
  The jersey number, as text.
</ResponseField>
<ResponseField name="position" type="string">
  The position, such as `QB` or `DE`.
</ResponseField>
<ResponseField name="college" type="string">
  The college the player came from.
</ResponseField>
<ResponseField name="birthPlace" type="string">
  Where the player was born.
</ResponseField>
<ResponseField name="birthDate" type="string">
  The birth date, as `YYYY-MM-DD`.
</ResponseField>
<ResponseField name="headshots" type="object">
  Photo links in `original`, `large`, `medium` & `small`. Any of them can be null.
</ResponseField>
<ResponseField name="xUser" type="object">
  The player's X account, with `id`, `username` & `blueVerified`. Null when the player has none.
</ResponseField>
