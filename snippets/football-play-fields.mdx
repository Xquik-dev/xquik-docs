<ResponseField name="id" type="string">
  X's ID for the play within the game.
</ResponseField>
<ResponseField name="sequence" type="string">
  The play's place in its drive, as X writes it.
</ResponseField>
<ResponseField name="clock" type="string">
  The game clock when the play started, as `HH:MM:SS`.
</ResponseField>
<ResponseField name="type" type="string">
  X's kind of play, such as `Pass`, `Rush`, `Punt` or `Conversion`.
</ResponseField>
<ResponseField name="direction" type="string">
  Where the play went, such as `Left` or `Middle`.
</ResponseField>
<ResponseField name="description" type="string">
  The play as the official play-by-play writes it.
</ResponseField>
<ResponseField name="scoring" type="boolean">
  Whether the play scored points.
</ResponseField>
<ResponseField name="homePoints" type="number">
  The home team's points after the play.
</ResponseField>
<ResponseField name="awayPoints" type="number">
  The away team's points after the play.
</ResponseField>
<ResponseField name="score" type="object">
  For a scoring play, its `points` & the `homePoints` & `awayPoints` after it. Null otherwise.
</ResponseField>
<ResponseField name="start" type="object">

Down & distance when the play started, or null.

<ResponseField name="down" type="number">
  The down, from 1 to 4.
</ResponseField>
<ResponseField name="yardsToGo" type="number">
  The yards needed for a first down.
</ResponseField>
<ResponseField name="yardline" type="number">
  The yard line of the ball.
</ResponseField>
<ResponseField name="yardlineSide" type="string">
  Whose half the ball was in. `Home` or `Away` in NFL games, the team's `alternateId` in college
  games.
</ResponseField>
<ResponseField name="possession" type="string">
  The team with the ball. `Home` or `Away` in NFL games, the team's `alternateId` in college games.
</ResponseField>

</ResponseField>
<ResponseField name="onField" type="object">
  Flags X shows on the field, or null. `calledBack` means a penalty called the play back.
  `penaltyOnField` means a flag was thrown. `timeout` means a timeout followed. `underReview` means
  officials were reviewing the play.
</ResponseField>
<ResponseField name="details" type="object[]">

The play's steps, in order, such as a pass, a tackle or a penalty.

<ResponseField name="sequence" type="number">
  The step's place in the play, from 1.
</ResponseField>
<ResponseField name="category" type="string">
  X's kind of step, such as `Pass`, `Tackle` or `Penalty`.
</ResponseField>
<ResponseField name="description" type="string">
  What happened in the step.
</ResponseField>
<ResponseField name="direction" type="string">
  The direction X gives for the step.
</ResponseField>
<ResponseField name="yards" type="number">
  The yards the step gained or lost.
</ResponseField>
<ResponseField name="players" type="object[]">
  The players in the step. Each has `playerId`, a key of `players`, & `playerRole`, such as `Pass`,
  `Receive` or `Tackle`.
</ResponseField>
<ResponseField name="penalty" type="object">
  The penalty this step called, or null. It has `description`, `nextDown`, `result`, `yards` &
  `teamId`, the penalized team.
</ResponseField>

</ResponseField>
<ResponseField name="post" type="object">
  The post X links to the play, with `id` & the author's `username`. Null when X links none.
</ResponseField>
<ResponseField name="videoPostId" type="string">
  The ID of a post with the play's video.
</ResponseField>
<ResponseField name="videoReplyPostId" type="string">
  The ID of a reply with the play's video.
</ResponseField>
