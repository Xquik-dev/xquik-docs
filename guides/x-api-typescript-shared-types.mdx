// The data field contains the tweet object stored with the monitor event.
interface TweetEventData {
  id: string;
  text: string;
  author: {
    id: string;
    userName: string;
    name: string;
  };
  isRetweet: boolean;
  isReply: boolean;
  isQuote: boolean;
  createdAt: string;
  inReplyToId?: string;
  quoted_tweet?: {
    id: string;
    text: string;
    author: { userName: string };
  };
}

type EventData = TweetEventData;

interface EventFilters {
  keywords?: { include?: string[]; exclude?: string[] };
  minLikes?: number;
  minRetweets?: number;
  verifiedOnly?: boolean;
}
